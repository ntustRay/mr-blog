筆記一下時間延遲的方法  
`Thread.Sleep` 只會暫停目前執行緒。此例的 `System.Timers.Timer` 預設在執行緒集區引發 Elapsed，因此主執行緒休息時仍可收到事件；這不表示每個 Timer 都有自己的專用執行緒，UI Timer 的行為也不同。

```csharp
using System;
using System.Threading; //thread要用
using System.Timers; // Timer要用
					
  public class Program{
        
        private static System.Timers.Timer aTimer; //宣告一個static Timer 
        
        public static void Main(String[] arg){
            
            Console.WriteLine("延遲前時間" + DateTime.Now.ToString("yyyy/MM/dd HH:mm:ss"));
            
            aTimer = new System.Timers.Timer(500); //約 0.5 秒後觸發
            aTimer.AutoReset = false; //此範例只執行一次

            aTimer.Elapsed += new ElapsedEventHandler(RUN);//執行RUN這個事件

            aTimer.Enabled = true; // 開始執行
		
            Thread.Sleep(2000); //執行續休息兩秒
            
            Console.WriteLine("Thread延遲" + DateTime.Now.ToString("yyyy/MM/dd HH:mm:ss"));
			
          }

      public static void RUN(Object source, ElapsedEventArgs e){
       // AutoReset=false 已設定成單次觸發。
       Console.WriteLine("Timer延遲" + DateTime.Now.ToString("yyyy/MM/dd HH:mm:ss"));
        }
    }
```
