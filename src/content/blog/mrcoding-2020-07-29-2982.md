```clike
public class CompareString {
    public static void main(String[] args) {
  
        String str1 = "Do you hello?";
        String str2 = new String(str1);
        String str3 = str1.toUpperCase();

        System.out.println(str1 == str2); // false 兩個變數參照不同的 String 物件
        System.out.println(str1.equals(str2));//true 比對物件裡的字串
        System.out.println(str1.equals(str3));//false 大小寫不一樣
        System.out.println(str1.equalsIgnoreCase(str3));//true 忽略大小寫         
    }  
}
```
