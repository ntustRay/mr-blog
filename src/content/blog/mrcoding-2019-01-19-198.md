```php
<?php

//資料庫設定
//資料庫位置
$db_server = "127.0.0.1";
//資料庫名稱
$db_name = "DBName";
//資料庫管理者帳號
$db_user = "root";
//資料庫管理者密碼
$db_passwd = "password";

//對資料庫連線
 $db = mysqli_connect($db_server, $db_user, $db_passwd, $db_name);
if ($db === false) {
    die("資料庫連線失敗");
}
      
//資料庫連線採UTF8
mysqli_set_charset($db, "utf8mb4");

//選擇資料庫
mysqli_select_db($db, $db_name)or die("無法使用資料庫");

// mysqli 是獨立的擴充套件與 API，不能把所有 mysql_* 函數只加 i 就當成完成遷移。
?> 
```

勘誤參考：[PHP：mysqli_set_charset](https://www.php.net/manual/en/mysqli.set-charset.php)。
