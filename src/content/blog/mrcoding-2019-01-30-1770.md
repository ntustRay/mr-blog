今天筆記一下

1. Array,List 命名方法
2. Array to List , List to Array 的方法
3. 還有各自長度不同的用法 length vs count

```
list.ToArray();  
list.Count ;

array.ToList();
array.Length ;
```

以下為範例

```clike
using System; 
using System.Collections.Generic; //系統集合泛型
using System.Linq; //Linq tolist

public class Program {
    public static void Main() {
        // 創建一個list
        List<string> str_list = new List<string>();
        str_list.Add("one"); //增加list 要用add
        str_list.Add("two");
        str_list.Add("three");
        str_list.Add("four");
        str_list.Add("five");
        str_list.Add(""); //故意加空字串
        str_list.Add("six");
        // list轉換成array 用ToArray()
        string[] array = str_list.ToArray(); 

        Console.WriteLine("Array的長度: " + array.Length + "\n" ); //取得array長度要用length
        foreach (var i in array) {  //看看有沒有正確列出所有的元素
            Console.WriteLine(i);
        }

        int[] int_array = new [] { 1,2,3,4,5,6 };
        // Array轉換成list 用ToList()
        List<int> int_list = int_array.ToList();

        Console.WriteLine("list的長度: " + int_list.Count);//取得list長度要用count
        foreach (var i in int_list) { //看看有沒有正確列出所有的元素
            Console.WriteLine(i);
        }
    }
}
```
