> 記得在 Inspector 指定 UI Text。targetFrameRate 是目標值，不保證實際效能；桌面／Web 開啟 VSync 時可能由 vSyncCount 主導。

今天來筆記一下做遊戲跟運作遊戲很需要的穩定幀數

```csharp
using UnityEngine;  
using System.Collections;  
using System.Collections.Generic;  
using UnityEngine.UI;

public class ShowFPS : MonoBehaviour {  

	public int fpsTarget = 60;  //通常設30 ~ 60
	public float updateInterval = 0.5f;  //每幾秒算一次
	private float lastInterval;  
	private int frames = 0;  
	private float fps;  
	public Text FPS_text; //讓UITEXT放進來


	void Start()  
	{  	  
		Application.targetFrameRate = fpsTarget;  //設定目標幀率，不能保證固定幀率
		lastInterval = Time.realtimeSinceStartup;  //自遊戲開始時間
		frames = 0;  //初始frames =0
	}  

	// 每一幀都會呼叫update()
	void Update()  
	{  
		frames++;   
		float timeNow = Time.realtimeSinceStartup;  
		if (timeNow >= lastInterval + updateInterval)  //每0.5秒更新一次
		{  
			fps = frames / (timeNow - lastInterval); //平均 FPS = 這段期間的畫格數 / 經過秒數 
			frames = 0;  
			lastInterval = timeNow;  
		}  

	}  

	void OnGUI()  
	{  
		FPS_text.text =  "FPS: " + fps.ToString() ; //在UI上顯示幀數
	}  
}  
```

勘誤參考：[Unity：Application.targetFrameRate](https://docs.unity3d.com/ScriptReference/Application-targetFrameRate.html)。
